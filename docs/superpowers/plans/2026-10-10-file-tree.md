# file-tree Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `file-tree`, a Claude Code mod: an IDE-style file tree in a pane with Claude-activity highlights, git status, search, a read-only preview, and three switchable styles.

**Architecture:** Pure modules (`tree.ts`, `git.ts`, `activity.ts`, `preview.ts`, `settings.ts`) hold all logic and are unit-tested. Three style modules turn a shared `RowInput` into `RowSpec` segments. One `Client` surface module (`rows.client.tsx`) draws the rows with the shimmer animation and posts keys and clicks back. `register.tsx` wires hooks, state atoms, the `/tree` command, the band button and the pane.

**Tech Stack:** Claude Code function hooks (plugin API of Claude Code 2.1.291), TypeScript/TSX compiled by the engine, `claude-code/testing` for tests, `git` and `find` through `$.process.run`.

**Spec:** `docs/superpowers/specs/2026-10-10-file-tree-design.md`

```ts
export const STYLE_NAMES = ['ide', 'minimal', 'rich'] as const
export type StyleName = (typeof STYLE_NAMES)[number]

export type Settings = {
  style: StyleName
  layout: 'replace' | 'split'
  openWith: string
  follow: boolean
  icons: 'nerd' | 'unicode'
  showIgnored: boolean
}

/**
 * Whether `value` names one of the styles.
 */
export function isStyle(value: unknown): value is StyleName {
  return typeof value === 'string' && (STYLE_NAMES as readonly string[]).includes(value)
}

/**
 * The settings from the plugin's `userConfig` options, defaults filled in.
 */
export function settingsOf(options: Record<string, unknown> | undefined): Settings {
  const o = options ?? {}

  return {
    style: isStyle(o.style) ? o.style : 'ide',
    layout: o.previewLayout === 'split' ? 'split' : 'replace',
    openWith: typeof o.openWith === 'string' ? o.openWith : '',
    follow: o.followClaude !== 'off',
    icons: o.icons === 'nerd' ? 'nerd' : 'unicode',
    showIgnored: o.showIgnored === 'on',
  }
}

/**
 * The style after `style` in the ◐ button's cycle.
 */
export function nextStyle(style: StyleName): StyleName {
  return STYLE_NAMES[(STYLE_NAMES.indexOf(style) + 1) % STYLE_NAMES.length] ?? 'ide'
}

/**
 * The style to draw: the ◐ button's stored choice while /config still holds
 * the value it overrode, else /config's.
 *
 * @param config the style /config holds
 * @param stored what the button stored: `{ style, over }`
 */
export function effectiveStyle(config: StyleName, stored: unknown): StyleName {
  const o = stored && typeof stored === 'object' ? (stored as { style?: unknown; over?: unknown }) : undefined

  return o && o.over === config && isStyle(o.style) ? o.style : config
}
```

```ts
/** One entry of a listed folder. */
export type Entry = { name: string; kind: 'file' | 'dir'; mtimeMs: number; isIgnored: boolean }

/** Every listed folder's sorted entries, by the folder's absolute path. */
export type Listing = Record<string, Entry[]>

/** One visible row of the tree. */
export type TreeRow = {
  path: string
  name: string
  depth: number
  kind: 'file' | 'dir'
  isOpen: boolean
  isIgnored: boolean
  mtimeMs: number
}

const HIDDEN = new Set(['.git', 'node_modules'])
const byName = (a: string, b: string) => a.localeCompare(b, undefined, { sensitivity: 'base' })

/**
 * `dir` and `name` joined with one slash.
 */
export function join(dir: string, name: string): string {
  return dir.endsWith('/') ? `${dir}${name}` : `${dir}/${name}`
}

/**
 * The folder holding `path`; `/` for a top-level path.
 */
export function parentOf(path: string): string {
  const at = path.lastIndexOf('/')

  return at <= 0 ? '/' : path.slice(0, at)
}

/**
 * Whether `path` is `root` or under it.
 */
export function isInside(root: string, path: string): boolean {
  return path === root || path.startsWith(root.endsWith('/') ? root : `${root}/`)
}

/**
 * `path` relative to `root` (no leading slash); `path` itself when outside.
 */
export function relative(root: string, path: string): string {
  if (path === root) return ''

  return isInside(root, path) ? path.slice(root.length).replace(/^\/+/, '') : path
}

/**
 * A folder's entries for the tree: files and folders only, `.git` and
 * `node_modules` left out, folders first, then by name ignoring case.
 *
 * @param ignored names git ignores in this folder
 * @param keepIgnored keep them, flagged, rather than leaving them out
 */
export function sortEntries(
  entries: readonly { name: string; kind: string; mtimeMs: number }[],
  ignored: ReadonlySet<string>,
  keepIgnored: boolean,
): Entry[] {
  return entries
    .filter(e => (e.kind === 'file' || e.kind === 'dir') && !HIDDEN.has(e.name))
    .filter(e => keepIgnored || !ignored.has(e.name))
    .map(e => ({ name: e.name, kind: e.kind as Entry['kind'], mtimeMs: e.mtimeMs, isIgnored: ignored.has(e.name) }))
    .sort((a, b) => (a.kind === b.kind ? byName(a.name, b.name) : a.kind === 'dir' ? -1 : 1))
}

/**
 * The visible rows: the root's entries, and under each open folder its own.
 */
export function flatten(root: string, listing: Listing, open: readonly string[]): TreeRow[] {
  const isOpen = new Set(open)
  const rows: TreeRow[] = []
  const walk = (dir: string, depth: number) => {
    for (const entry of listing[dir] ?? []) {
      const path = join(dir, entry.name)
      const opened = entry.kind === 'dir' && isOpen.has(path)
      rows.push({ path, name: entry.name, depth, kind: entry.kind, isOpen: opened, isIgnored: entry.isIgnored, mtimeMs: entry.mtimeMs })
      if (opened) walk(path, depth + 1)
    }
  }
  walk(root, 0)

  return rows
}

/**
 * The folders to open, outermost first, so that `path` shows in the tree.
 */
export function foldersTo(root: string, path: string): string[] {
  if (path === root || !isInside(root, path)) return []
  const out: string[] = []
  for (let dir = parentOf(path); dir !== root && isInside(root, dir); dir = parentOf(dir)) out.unshift(dir)

  return out
}

/**
 * Lowercase text with diacritics removed, so `y tuong` finds `Ý tưởng`.
 */
export function fold(text: string): string {
  return text.normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/đ/gi, 'd').toLowerCase()
}

/**
 * Whether every word of `query` is in `path`, ignoring case and diacritics.
 */
export function matches(path: string, query: string): boolean {
  const haystack = fold(path)

  return fold(query).split(/\s+/).filter(Boolean).every(word => haystack.includes(word))
}

type SearchNode = { dirs: Map<string, SearchNode>; files: string[] }

/**
 * The rows of a search: the matching files (paths relative to `root`) as a
 * tree opened down to each, folders first, at most `limit` files.
 */
export function searchRows(root: string, files: readonly string[], query: string, limit = 200): TreeRow[] {
  const top: SearchNode = { dirs: new Map(), files: [] }
  for (const hit of files.filter(file => matches(file, query)).slice(0, limit)) {
    const parts = hit.split('/')
    let node = top
    for (const part of parts.slice(0, -1)) {
      const next = node.dirs.get(part) ?? { dirs: new Map(), files: [] }
      node.dirs.set(part, next)
      node = next
    }
    node.files.push(parts.at(-1) ?? hit)
  }
  const rows: TreeRow[] = []
  const walk = (node: SearchNode, dir: string, depth: number) => {
    for (const name of [...node.dirs.keys()].sort(byName)) {
      const path = join(dir, name)
      rows.push({ path, name, depth, kind: 'dir', isOpen: true, isIgnored: false, mtimeMs: 0 })
      const child = node.dirs.get(name)
      if (child) walk(child, path, depth + 1)
    }
    for (const name of [...node.files].sort(byName)) {
      rows.push({ path: join(dir, name), name, depth, kind: 'file', isOpen: false, isIgnored: false, mtimeMs: 0 })
    }
  }
  walk(top, root, 0)

  return rows
}
```

```ts
import { isInside, join, parentOf } from './tree'

export type GitStatus = 'M' | 'A' | 'U' | 'D' | 'R' | '!'
export type FileGit = { status: GitStatus; added?: number; removed?: number }
export type Counts = { M: number; U: number; D: number }
export type GitSnapshot = {
  branch: string
  upstream: string
  ahead: number
  behind: number
  files: Record<string, FileGit>
}

/**
 * A file's status from porcelain v2's `XY`: deleted, added, else modified.
 */
function xyStatus(xy: string): GitStatus {
  if (xy.includes('D')) return 'D'
  if (xy.startsWith('A')) return 'A'

  return 'M'
}

/**
 * Reads `git status --porcelain=v2 --branch -z`: the branch line facts and a
 * status per changed path, keyed by absolute path under `top`.
 */
export function parseStatus(out: string, top: string): GitSnapshot {
  const snap: GitSnapshot = { branch: '', upstream: '', ahead: 0, behind: 0, files: {} }
  const fields = out.split('\0')
  const put = (path: string, status: GitStatus) => {
    snap.files[join(top, path.replace(/\/+$/, ''))] = { status }
  }
  for (let i = 0; i < fields.length; i++) {
    const field = fields[i] ?? ''
    if (field.startsWith('# branch.head ')) snap.branch = field.slice('# branch.head '.length)
    else if (field.startsWith('# branch.upstream ')) snap.upstream = field.slice('# branch.upstream '.length)
    else if (field.startsWith('# branch.ab ')) {
      const [ahead, behind] = field.slice('# branch.ab '.length).split(' ')
      snap.ahead = Math.abs(Number(ahead) || 0)
      snap.behind = Math.abs(Number(behind) || 0)
    } else if (field.startsWith('1 ')) put(field.split(' ').slice(8).join(' '), xyStatus(field.split(' ')[1] ?? ''))
    else if (field.startsWith('2 ')) {
      put(field.split(' ').slice(9).join(' '), 'R')
      i++
    } else if (field.startsWith('u ')) put(field.split(' ').slice(10).join(' '), '!')
    else if (field.startsWith('? ')) put(field.slice(2), 'U')
  }

  return snap
}

/**
 * Reads `git diff --numstat -z HEAD`: added and removed lines per path, keyed
 * by absolute path under `top`; a binary file counts as 0 and 0.
 */
export function parseNumstat(out: string, top: string): Record<string, { added: number; removed: number }> {
  const stats: Record<string, { added: number; removed: number }> = {}
  const fields = out.split('\0')
  for (let i = 0; i < fields.length; i++) {
    const m = /^(\d+|-)\t(\d+|-)\t([\s\S]*)$/.exec(fields[i] ?? '')
    if (!m) continue
    let path = m[3] ?? ''
    if (path === '') {
      path = fields[i + 2] ?? ''
      i += 2
    }
    stats[join(top, path)] = { added: Number(m[1]) || 0, removed: Number(m[2]) || 0 }
  }

  return stats
}

/**
 * The snapshot with each file's line counts filled in.
 */
export function withNumstat(snap: GitSnapshot, stats: Record<string, { added: number; removed: number }>): GitSnapshot {
  const files: Record<string, FileGit> = {}
  for (const [path, git] of Object.entries(snap.files)) files[path] = { ...git, ...(stats[path] ?? {}) }

  return { ...snap, files }
}

/**
 * Per folder under `root` (itself included), how many changed files it holds:
 * modified (renamed and conflicted too), untracked or added, deleted.
 */
export function rollUp(files: Record<string, FileGit>, root: string): Record<string, Counts> {
  const counts: Record<string, Counts> = {}
  for (const [path, git] of Object.entries(files)) {
    if (!isInside(root, path) || path === root) continue
    const bucket: keyof Counts = git.status === 'U' || git.status === 'A' ? 'U' : git.status === 'D' ? 'D' : 'M'
    for (let dir = parentOf(path); isInside(root, dir); dir = parentOf(dir)) {
      const c = (counts[dir] ??= { M: 0, U: 0, D: 0 })
      c[bucket]++
      if (dir === root) break
    }
  }

  return counts
}
```

```ts
import { join, relative } from './tree'

export type Kind = 'read' | 'write' | 'commit'
export type Mark = { path: string; kind: Kind; isLive: boolean; until: number }

/** How long a mark stays tinted after its tool call ended. */
export const FADE_MS = 4000

const READ_KEYS: Record<string, string> = { Read: 'file_path', Grep: 'path', Glob: 'path' }
const WRITE_KEYS: Record<string, string> = { Edit: 'file_path', Write: 'file_path', NotebookEdit: 'notebook_path' }

/**
 * `path` made absolute against `cwd`, without a trailing slash.
 */
export function absolute(cwd: string, path: string): string {
  const full = path.startsWith('/') ? path : join(cwd, path.replace(/^\.\//, ''))

  return full.replace(/\/+$/, '') || '/'
}

/**
 * The file or folder a tool call reads or writes, or null for any other tool.
 * Grep and Glob with no path search the working directory.
 */
export function targetOf(input: { tool: string } & Record<string, unknown>, cwd: string): { kind: Kind; path: string } | null {
  const readKey = READ_KEYS[input.tool]
  const writeKey = WRITE_KEYS[input.tool]
  const key = readKey ?? writeKey
  if (!key) return null
  const raw = input[key]
  const path = typeof raw === 'string' && raw !== '' ? raw : input.tool === 'Grep' || input.tool === 'Glob' ? cwd : ''
  if (!path) return null

  return { kind: writeKey ? 'write' : 'read', path: absolute(cwd, path) }
}

/**
 * Whether a shell command line runs `git commit` (as a command, not as text).
 */
export function isCommit(command: string): boolean {
  return /(^|[;&|(]\s*)git(\s+-C\s+\S+)*\s+commit(\s|$)/.test(command)
}

/**
 * The marks with a live one on each of `paths`, replacing marks on them.
 */
export function startMarks(marks: readonly Mark[], paths: readonly string[], kind: Kind): Mark[] {
  return [...marks.filter(m => !paths.includes(m.path)), ...paths.map(path => ({ path, kind, isLive: true, until: 0 }))]
}

/**
 * The marks with those on `paths` turned to fading, until `now + FADE_MS`.
 */
export function endMarks(marks: readonly Mark[], paths: readonly string[], now: number): Mark[] {
  return marks.map(m => (m.isLive && paths.includes(m.path) ? { ...m, isLive: false, until: now + FADE_MS } : m))
}

/**
 * The marks still showing at `now`.
 */
export function pruneMarks(marks: readonly Mark[], now: number): Mark[] {
  return marks.filter(m => m.isLive || m.until > now)
}

const VERB: Record<Kind, string> = { read: 'reading', write: 'editing', commit: 'committing' }

/**
 * The rich style's header line: what Claude is doing to which file now, or ''.
 */
export function statusLine(marks: readonly Mark[], root: string): string {
  const live = marks.filter(m => m.isLive).at(-1)

  return live ? `Claude is ${VERB[live.kind]} ${relative(root, live.path) || '.'}` : ''
}
```

```ts
import { parentOf } from './tree'

/** The most text one preview holds (the Code element draws 100000 at most). */
export const MAX_CHARS = 90_000

/** Files larger than this are not read (`$.fs.read` refuses over 4 MiB). */
export const MAX_BYTES = 4_000_000

/**
 * Whether the file is drawn as Markdown rather than as code.
 */
export function isMarkdown(path: string): boolean {
  return /\.(md|mdx|markdown)$/i.test(path)
}

/**
 * Whether text holds a NUL byte: a binary file, not shown.
 */
export function isBinary(text: string): boolean {
  return text.includes('\u0000')
}

/**
 * Text cut to whole lines under MAX_CHARS, with a note when cut.
 */
export function clip(text: string): { text: string; note: string } {
  if (text.length <= MAX_CHARS) return { text, note: '' }
  const cut = text.slice(0, MAX_CHARS)
  const end = cut.lastIndexOf('\n')
  const kept = end > 0 ? cut.slice(0, end) : cut

  return { text: kept, note: `first ${kept.split('\n').length} lines` }
}

/**
 * The highest first line a window of `rows` can start at.
 */
export function maxTop(text: string, rows: number): number {
  return Math.max(0, text.split('\n').length - rows)
}

/**
 * The `rows` lines from line `top` (0-based, clamped), and the 1-based number
 * of the first.
 */
export function windowOf(text: string, top: number, rows: number): { source: string; startLine: number } {
  const lines = text.split('\n')
  const from = Math.max(0, Math.min(top, lines.length - rows))

  return { source: lines.slice(from, from + rows).join('\n'), startLine: from + 1 }
}

/**
 * A command line split on spaces, single or double quotes keeping a word whole.
 */
export function splitWords(command: string): string[] {
  return [...command.matchAll(/"([^"]*)"|'([^']*)'|(\S+)/g)].map(m => m[1] ?? m[2] ?? m[3] ?? '')
}

/**
 * What the ↗ button runs: the Open with command with `{path}` and `{dir}`
 * filled (the path last when it holds no `{path}`), or the platform's opener.
 */
export function openArgv(setting: string, path: string, os: 'darwin' | 'linux' | 'win32'): string[] {
  if (!setting.trim()) {
    if (os === 'darwin') return ['open', '--', path]

    return os === 'win32' ? ['cmd', '/c', 'start', '', path] : ['xdg-open', path]
  }
  const words = splitWords(setting)
  const argv = words.map(w => w.replaceAll('{path}', path).replaceAll('{dir}', parentOf(path)))

  return words.some(w => w.includes('{path}')) ? argv : [...argv, path]
}
```

```ts
import type { Counts, FileGit, GitSnapshot, GitStatus } from './git'
import type { Kind, Mark } from './activity'
import type { TreeRow } from './tree'

/** One run of text in a row: color (a theme key or hex), and how it lights. */
export type Seg = {
  t: string
  c?: string
  b?: boolean
  dim?: boolean
  /** Claude's activity tone; with `sweep` a light band runs across the text. */
  tone?: Kind
  sweep?: boolean
  /** Shown on alternate animation frames. */
  blink?: boolean
}

/** One drawn row: `id` is the path a click reports ('' for no click). */
export type RowSpec = { id: string; left: Seg[]; right: Seg[] }

/** What a style draws a row from. */
export type RowInput = TreeRow & { git?: FileGit; counts?: Counts; mark?: Mark }

/** What a style draws the header from. */
export type HeaderInput = { name: string; git: GitSnapshot | null; dirty: boolean; status: string }

export type StyleCtx = { icons: 'nerd' | 'unicode'; now: number }

export type Style = {
  row: (r: RowInput, ctx: StyleCtx) => RowSpec
  header: (h: HeaderInput, ctx: StyleCtx) => RowSpec[]
}

/** Theme keys per git status, so colors follow the person's theme. */
export const GIT_COLOR: Record<GitStatus, string> = { M: 'warning', A: 'success', U: 'success', D: 'error', R: 'suggestion', '!': 'error' }

/**
 * The status a folder shows: the first of modified, untracked, deleted it holds.
 */
export function folderStatus(counts: Counts | undefined): GitStatus | undefined {
  if (!counts) return undefined
  if (counts.M > 0) return 'M'
  if (counts.U > 0) return 'U'

  return counts.D > 0 ? 'D' : undefined
}

/**
 * The row's name segment: dim when ignored, colored by git, lit by a mark
 * (sweeping while live when `sweep` is asked for, tinted while fading).
 */
export function nameSeg(r: RowInput, sweep: boolean, base: Seg = {}): Seg {
  const status = r.kind === 'dir' ? folderStatus(r.counts) : r.git?.status
  const seg: Seg = { ...base, t: r.name, ...(status ? { c: GIT_COLOR[status] } : {}), ...(r.isIgnored ? { dim: true } : {}) }
  if (!r.mark) return seg

  return { ...seg, tone: r.mark.kind, ...(r.mark.isLive && sweep ? { sweep: true } : {}) }
}

/**
 * How long ago `ms` was, in the largest whole unit: `5m`, `3h`, `2d`, `4w`.
 */
export function age(ms: number, now: number): string {
  const minutes = Math.max(0, Math.round((now - ms) / 60000))
  if (minutes < 60) return `${minutes}m`
  if (minutes < 60 * 24) return `${Math.round(minutes / 60)}h`
  if (minutes < 60 * 24 * 14) return `${Math.round(minutes / 60 / 24)}d`

  return `${Math.round(minutes / 60 / 24 / 7)}w`
}

/**
 * The header's first line, shared by the styles: the root's name on the
 * left, branch, ahead/behind and a dot for uncommitted changes on the right.
 */
export function titleRow(h: HeaderInput, title: Seg[]): RowSpec {
  const right: Seg[] = h.git
    ? [
        { t: h.git.branch || '(detached)', c: 'suggestion' },
        ...(h.git.ahead ? [{ t: ` ↑${h.git.ahead}`, dim: true }] : []),
        ...(h.git.behind ? [{ t: ` ↓${h.git.behind}`, dim: true }] : []),
        ...(h.dirty ? [{ t: ' ●', c: 'warning' }] : []),
      ]
    : []

  return { id: '', left: title, right }
}
```

```tsx
import type { ClientModule } from 'claude-code'

import type { Kind } from './activity'
import type { RowSpec, Seg } from './segments'

export type RowsProps = {
  rows: RowSpec[]
  active: string
  preview?: { text: string; path: string; startLine: number; markdown: boolean }
}

type Local = { phase: number; hover: number; last: { id: string; at: number }; ref: { stop?: () => void } }

const TICK_MS = 90
const DOUBLE_MS = 400

/** Dark to bright, per activity tone. */
export const TONES: Record<Kind, string[]> = {
  read: ['#5b4a7a', '#8a6fc0', '#b89cf0', '#e6d8ff'],
  write: ['#7a4a2a', '#c0763a', '#f0a060', '#ffd8b0'],
  commit: ['#2f6a3a', '#4caf62', '#7fdc8f', '#c8f5d0'],
}

/**
 * The color of character `i` of a `len`-long name at animation `phase`: a
 * band of light that runs left to right and starts again.
 */
export function sweepColor(i: number, phase: number, len: number, palette: readonly string[]): string {
  const center = ((phase * 1.5) % (len + 8)) - 4
  const d = Math.abs(i - center)
  const step = d < 0.8 ? 3 : d < 1.8 ? 2 : d < 2.8 ? 1 : 0

  return palette[step] ?? palette[0] ?? '#ffffff'
}

const Rows: ClientModule<RowsProps, Local> = (props, surface) => {
  const { Box, Text, Code, Markdown } = surface.elements
  let state = surface.state
  if (state === undefined) {
    state = { phase: 0, hover: -1, last: { id: '', at: 0 }, ref: {} }
    surface.setState(state)
  }
  const animates = props.rows.some(r => [...r.left, ...r.right].some(s => s.sweep || s.blink))
  if (animates && !state.ref.stop) {
    state.ref.stop = surface.every(TICK_MS, () => {
      const cur = surface.state
      if (cur) surface.setState({ ...cur, phase: cur.phase + 1 })
    })
  } else if (!animates && state.ref.stop) {
    state.ref.stop()
    state.ref.stop = undefined
  }

  surface.onKey(e => surface.post({ key: e.key }))
  surface.onPointer(e => {
    const cur = surface.state ?? state
    if (!cur) return
    if (e.type === 'leave' || e.y < 0 || e.y >= props.rows.length) {
      if (cur.hover !== -1) surface.setState({ ...cur, hover: -1 })

      return
    }
    if (e.type === 'move' && cur.hover !== e.y) surface.setState({ ...cur, hover: e.y })
    if (e.type !== 'down' || (e.button ?? 'left') !== 'left') return
    const id = props.rows[e.y]?.id
    if (!id) return
    const now = Date.now()
    const isDouble = cur.last.id === id && now - cur.last.at < DOUBLE_MS
    surface.setState({ ...cur, last: { id, at: isDouble ? 0 : now } })
    surface.post(isDouble ? { open: id } : { press: id })
  })

  if (props.preview) {
    const p = props.preview

    return p.markdown
      ? <Markdown text={p.text} />
      : <Code source={p.text} path={p.path} startLine={p.startLine} wrap="truncate-end" />
  }

  const phase = state.phase
  const draw = (s: Seg) => {
    if (s.blink && phase % 6 >= 3) return <Text>{' '.repeat([...s.t].length)}</Text>
    const palette = s.tone ? TONES[s.tone] : undefined
    if (palette && s.sweep) {
      const chars = [...s.t]

      return (
        <Text bold={s.b}>
          {chars.map((ch, i) => <Text color={sweepColor(i, phase, chars.length, palette)}>{ch}</Text>)}
        </Text>
      )
    }
    const color = palette ? palette[2] : s.c

    return <Text color={color} bold={s.b} dimColor={s.dim}>{s.t}</Text>
  }

  return (
    <Box flexDirection="column">
      {props.rows.map((r, i) => (
        <Box
          flexDirection="row"
          height={1}
          overflow="hidden"
          backgroundColor={r.id && r.id === props.active ? '#313244' : i === state.hover ? '#262637' : undefined}
        >
          <Box flexShrink={1} overflow="hidden">{r.left.map(draw)}</Box>
          <Box flexGrow={1} />
          {r.right.map(draw)}
        </Box>
      ))}
    </Box>
  )
}

export default Rows
```

```tsx
/**
 * Lists one folder for the tree, leaving out (or flagging) what git ignores.
 */
async function listDir($: EngineInterface, dir: string): Promise<Entry[]> {
  const raw = await $.fs.list(dir)
  let ignored = new Set<string>()
  if (gitTop && raw.length) {
    const names = raw.map(e => (e.kind === 'dir' ? `${e.name}/` : e.name))
    const run = await $.process.run(['git', '--no-optional-locks', '-C', dir, 'check-ignore', '--', ...names], { timeoutMs: 5000 })
      .catch(() => ({ exitCode: 1, stdout: '' }))
    ignored = new Set(run.stdout.split('\n').filter(Boolean).map(p => p.replace(/\/$/, '').split('/').pop() ?? p))
  }

  return sortEntries(raw, ignored, settings.showIgnored)
}

/**
 * Lists `dirs` again (each one listed before, or asked for), keeping the rest.
 */
async function relist($: EngineInterface, dirs: readonly string[]) {
  const lists = await Promise.all(dirs.map(async dir => [dir, await listDir($, dir).catch(() => null)] as const))
  await update($, listing, all => {
    const next = { ...all }
    for (const [dir, entries] of lists) if (entries) next[dir] = entries

    return next
  })
}

/**
 * Re-roots the tree on the session's working directory: lists its top folder
 * and forgets what was open.
 */
async function reset($: EngineInterface) {
  const cwd = await $.session.cwd()
  const check = await $.process.run(['git', '-C', cwd, 'rev-parse', '--show-toplevel'], { timeoutMs: 5000 })
    .catch(() => ({ exitCode: 1, stdout: '' }))
  gitTop = check.exitCode === 0 ? check.stdout.trim() : null
  await update($, root, () => cwd)
  await update($, listing, () => ({}))
  await update($, open, () => [])
  await update($, cursor, () => '')
  await update($, top, () => 0)
  await update($, query, () => '')
  await update($, files, () => [])
  await relist($, [cwd])
}

/**
 * Opens or closes a folder, listing it the first time it opens.
 */
async function toggle($: EngineInterface, dir: string) {
  const isOpen = (await read($, open)).includes(dir)
  if (isOpen) {
    await update($, open, list => list.filter(d => d !== dir))

    return
  }
  if (!(await read($, listing))[dir]) await relist($, [dir])
  await update($, open, list => [...list, dir])
}

  const q = await read($, query)

  return q.trim() ? searchRows(r, await read($, files), q) : flatten(r, await read($, listing), await read($, open))
}

/**
 * Moves the cursor to `path` and scrolls so that it shows.
 */
async function moveTo($: EngineInterface, rows: readonly TreeRow[], path: string) {
  const at = rows.findIndex(row => row.path === path)
  if (at < 0) return
  await update($, cursor, () => path)
  await update($, top, t => (at < t ? at : at >= t + room ? at - room + 1 : t))
}

/**
 * Acts on a key the rows region sent: move, open, close, page.
 */
async function onKey($: EngineInterface, key: string) {
  const rows = await treeRows($)
  const cur = await read($, cursor)
  const at = Math.max(0, rows.findIndex(row => row.path === cur))
  const row = rows[at]
  const go = (delta: number) => {
    const target = rows[Math.max(0, Math.min(rows.length - 1, at + delta))]

    return target ? moveTo($, rows, target.path) : Promise.resolve()
  }
  if (key === 'up' || key === 'k') await go(-1)
  else if (key === 'down' || key === 'j') await go(1)
  else if (key === 'pageup') await go(-room)
  else if (key === 'pagedown') await go(room)
  else if (key === 'home') await go(-rows.length)
  else if (key === 'end') await go(rows.length)
  else if (!row) return
  else if ((key === 'right' || key === 'l') && row.kind === 'dir' && !row.isOpen) await toggle($, row.path)
  else if (key === 'left' || key === 'h') {
    if (row.kind === 'dir' && row.isOpen) await toggle($, row.path)
    else if (parentOf(row.path) !== (await read($, root))) await moveTo($, rows, parentOf(row.path))
  } else if ((key === 'return' || key === ' ') && row.kind === 'dir') await toggle($, row.path)
}

/**
 * The rows with their git status, folder counts and Claude's marks.

async function decorate($: EngineInterface, rows: readonly TreeRow[]): Promise<RowInput[]> {
```

```tsx
    if (st.size > MAX_BYTES) why = `${Math.round(st.size / 1e6)} MB, too large to preview`
    else {
      const raw = await $.fs.read(path)
      if (isBinary(raw)) why = 'binary file'
      else {
        const cut = clip(raw)
        text = cut.text
        why = cut.note
      }
    }
  }
  await update($, preview, () => path)
  await update($, previewText, () => text)
  await update($, previewNote, () => why)
  await update($, previewTop, () => 0)
}

/**
 * Closes the preview and returns to the tree.
 */
async function closePreview($: EngineInterface) {
  await update($, preview, () => '')
  await update($, previewText, () => '')
}

/**
 * Scrolls the preview by `by` lines, clamped.
 */
async function scrollPreview($: EngineInterface, by: number) {
  const last = maxTop(await read($, previewText), previewRoom)
  await update($, previewTop, t => Math.max(0, Math.min(last, t + by)))
}

/**
 * Opens the previewed file with the Open with command or the default app.
 */
async function openExternal($: EngineInterface, path: string) {
  if (!os) {
    const uname = await $.process.run(['uname', '-s'], { timeoutMs: 3000 }).catch(() => ({ exitCode: 1, stdout: '' }))
    os = uname.exitCode !== 0 ? 'win32' : uname.stdout.trim() === 'Darwin' ? 'darwin' : 'linux'
  }
  const run = await $.process.run(openArgv(settings.openWith, path, os), { timeoutMs: 10000 })
    .catch((error: unknown) => ({ exitCode: 1, stdout: '', stderr: String(error) }))
  if (run.exitCode !== 0) await update($, note, () => `Could not open ${path.split('/').pop()}: ${('stderr' in run ? run.stderr : '').trim().slice(0, 80)}`)
}
```
